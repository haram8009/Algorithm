import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Solution {

	static int[] ticketPrices;
	static int[] plans;

	static int[] dp;

	public static void main(String[] args) throws IOException {

		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());

		int t = Integer.parseInt(st.nextToken());

		for (int tc = 1; tc <= t; tc++) {
			ticketPrices = new int[4]; // 1일, 1달, 3달, 1년
			plans = new int[13];

			st = new StringTokenizer(br.readLine(), " ");
			for (int i = 0; i < ticketPrices.length; i++) {
				ticketPrices[i] = Integer.parseInt(st.nextToken());
			}

			st = new StringTokenizer(br.readLine(), " ");
			for (int i = 1; i < plans.length; i++) {
				plans[i] = Integer.parseInt(st.nextToken());
			}

			dp = new int[13]; // 매달 최적 구매 비용

			// 일일, 월, 3개월
			dp[0] = 0;
			for (int i = 1; i < dp.length; i++) {
				dp[i] = dp[i - 1] + Math.min(plans[i] * ticketPrices[0], ticketPrices[1]);
				if (i >= 3) {
					dp[i] = Math.min(dp[i], dp[i - 3] + ticketPrices[2]);
				}
			}

			System.out.println("#" + tc + " " + Math.min(dp[12], ticketPrices[3]));
		}
	}

}


// 시간복잡도: O(N)

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Collections;
import java.util.StringTokenizer;

class Solution {
	static int n, ans;
	static int[][] map;
	static int[][] people; // [i] = {행, 열}
	static int[][] stair; // [s] = {행, 열, 길이}
	static int pCnt;
	static int[] choice; // choice[i] = 사람 i가 선택한 계단(0 또는 1)

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int t = Integer.parseInt(st.nextToken());

		for (int tc = 1; tc <= t; tc++) {
			ans = Integer.MAX_VALUE;
			st = new StringTokenizer(br.readLine());
			n = Integer.parseInt(st.nextToken());
			map = new int[n][n];

			people = new int[10][2];
			stair = new int[2][3];
			choice = new int[10];
			pCnt = 0;
			int idx = 0;
			for (int i = 0; i < n; i++) {
				st = new StringTokenizer(br.readLine(), " ");
				for (int j = 0; j < n; j++) {
					map[i][j] = Integer.parseInt(st.nextToken());
					if (map[i][j] == 1)
						people[pCnt++] = new int[] { i, j };
					else if (map[i][j] > 1)
						stair[idx++] = new int[] { i, j, map[i][j] };
				}
			}

//			for (int[] r : map) {
//				System.out.println(Arrays.toString(r));
//			}

			// 각 계단에 누가 배정될지 부분집합 -> 모든 경우의 수 시뮬레이션 해보기
			// (계단 2개, 사람수 10, 방 크기 10x10 -> 완탐 ㄱㄴ
			// 가지치기 가능 (최종 시간 이전에 기록한 시간보다 길어지면 다음경우 시뮬)
			choice(0);

			System.out.println("#" + tc + " " + ans);

		}

	}

	// 계단 선택
	static void choice(int depth) {
		if (depth == pCnt) {
			calc();
			return;
		}
		choice[depth] = 1;
		choice(depth + 1);
		choice[depth] = 0;
		choice(depth + 1);

	}

	// 시뮬레이션, 계산
	static void calc() {
		int totalTime = 0;

		// 두 계단에 대해 배정된 사람 모두 이동하는 시간 구하고 더 긴게 해당 배정의 시간
		for (int s = 0; s < 2; s++) {
			int thisTime = 0;
			int k = stair[s][2];
			int sCnt = 0;
			int[] t = new int[10];
			Arrays.fill(t, Integer.MAX_VALUE);
			for (int i = 0; i < pCnt; i++) {
				if (choice[i] == s) {
					int dist = Math.abs(people[i][0] - stair[s][0]) + Math.abs(people[i][1] - stair[s][1]);
					t[sCnt++] = dist;
				}
			}
			if (sCnt == 0)
				continue;
			Arrays.sort(t);
			for (int i = 0; i < sCnt; i++) {
				int start = t[i] + 1; // 도착 + 1분 대기
				if (i >= 3)
					start = Math.max(start, t[i - 3]); // t[i-3]은 이미 완료 시각
				t[i] = start + k; // 완료 시각
			}
			thisTime = t[sCnt - 1]; // 마지막 완주한 사람의 완주시각

			totalTime = Math.max(totalTime, thisTime);
		}

		ans = Math.min(ans, totalTime);
	}

}

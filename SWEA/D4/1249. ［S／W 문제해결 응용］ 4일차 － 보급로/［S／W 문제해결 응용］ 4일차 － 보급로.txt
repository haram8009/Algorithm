import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Solution {
	static int n;

	static int[][] map;
	static int[][] dp;
	static boolean[][] visited;

	// 상 하 좌 우
	static int[] dr = { -1, 1, 0, 0 };
	static int[] dc = { 0, 0, -1, 1 };

	static class Point implements Comparable<Point> {
		int r;
		int c;
		int w;

		public Point(int r, int c, int w) {
			super();
			this.r = r;
			this.c = c;
			this.w = w;
		}

		@Override
		public int compareTo(Point o) {
			return Integer.compare(this.w, o.w);
		}
	}

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int t = Integer.parseInt(st.nextToken());

		for (int tc = 1; tc <= t; tc++) {
			st = new StringTokenizer(br.readLine());

			n = Integer.parseInt(st.nextToken());

			map = new int[n][n];
			dp = new int[n][n];
			visited = new boolean[n][n];

			for (int i = 0; i < n; i++) {
//				st = new StringTokenizer(br.readLine(), "");
				char[] line = br.readLine().toCharArray();
				for (int j = 0; j < n; j++) {
					map[i][j] = line[j] - '0';
					dp[i][j] = Integer.MAX_VALUE;
				}
			}

//			for (int i = 0; i < n; i++) {
//				System.out.println(Arrays.toString(map[i]));
//			}

			PriorityQueue<Point> pq = new PriorityQueue<>();
			dp[0][0] = 0;
			pq.offer(new Point(0, 0, 0));

			while (!pq.isEmpty()) {
				Point pos = pq.poll();
//				System.out.println("===\npos r, c: " + pos.r + ", " + pos.c);
				if (visited[pos.r][pos.c])
					continue;
				if(pos.w > dp[pos.r][pos.c])
					continue;

				visited[pos.r][pos.c] = true;

				for (int d = 0; d < 4; d++) {
					int nr = pos.r + dr[d];
					int nc = pos.c + dc[d];
					if (isInRange(nr, nc)) {
//						System.out.println("- nr, nc: " + nr + ", " + nc);

						int dist = dp[pos.r][pos.c] + map[nr][nc];

						if (!visited[nr][nc] && dist < dp[nr][nc]) {
							dp[nr][nc] = dist;
							pq.offer(new Point(nr, nc, dist));
						}

//						System.out.println("dp[nr][nc]: " + dist);							
					}
				}
			}

			System.out.println("#" + tc + " " + dp[n - 1][n - 1]);
		}
	}

	static boolean isInRange(int r, int c) {
		return r >= 0 && c >= 0 && r < n && c < n;
	}
}
